void activate_watering_system(uint16_t duration) {
	// Activate LED0 indicating watering system is active
	PORTD.OUTSET = PIN0_bm; // after the notification which will still stand for not having misunderstanding
	// Start timer/counter for the calculated duration
	TCA0.SINGLE.PER = duration; // Set period
	TCA0.SINGLE.CTRLA = TCA_SINGLE_ENABLE_bm | TCA_SINGLE_CLKSEL_DIV1_gc;
	cli();
	TCA0_hardReset(); //after the watering system gets activated the system gets to the initial state
}




void activate_ventilation_system() {
	// Enable split mode
	TCA0.SPLIT.CTRLD = TCA_SPLIT_SPLITM_bm;
	// Activate LED1 indicating ventilation system is active
	PORTD.OUTSET = PIN1_bm;
	// Configure PWM for LED2 (PORTD pin 2)
	TCB1.CTRLA = TCB_ENABLE_bm; // Enable TCB1
	TCB1.CTRLB = TCB_CCMPEN_bm; // Enable compare mode
	TCB1.CCMP = 0x3FF; // Set compare value for 50% duty cycle
	TCB1.CTRLB |= TCB_CNTMODE_PWM8_gc; // Set PWM mode
	TCB1.CTRLA |= TCB_ENABLE_bm; // Enable PWM output
	cli();
	//TCA0_hardReset();
}
