#include <avr/io.h>
#include <avr/interrupt.h>
#include <util/delay.h>


#define T1 10000   //Tram comes and interrupts normal operation  (in milliseconds)
#define T2 3000   // Interruption of Tram where RED is for Cars and GREEN for Pedestrians AND in the other case the pedestrian button is pressed where GREEN is for Pedestrians and RED for Cars (in milliseconds)
#define T3 5000   // Time interval for preventing immediate pedestrian button press (in milliseconds)

int x=0; //logic flag

int main() {
	
	PORTD.DIR |= 0b00000010; //PIN is output
	PORTD.OUT= 0b00000010;//LED is off
	
	 sei(); //enable interrupts

	while (x==0) {
		
	 PORTD.OUTCLR= 0b00000010;//PIN1_bm //LED is on
	 _delay_ms(T1); //wait for 10 ms
	
	 cli(); //disable interrupts

	}
}

ISR(PORTD_PORT_vect){
	int y = PORTD.INTFLAGS; //Procedure to
	PORTD.INTFLAGS=y; //clear the interrupt flag
	x=1; //change logic flag to get out of loop
}
