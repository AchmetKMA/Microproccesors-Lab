// Initialize TCA0 in split mode
void initTCA0() {
	
	
	TCA0.SPLIT = TCA_SPLIT_SPLITM_bm; // Enable split mode
	TCA0.SPLIT.CTRLA |= TCA_SPLIT_ENABLE_bm; // Enable TCA0
	TCA0.SPLIT.HCMP0 = 0; //clear counter
	TCA0.SPLIT.LCMP0 = 0; //clear counter
	
	
	TCA0.SPLIT.HCMP0 = T1_count_value;      // Set Counter A value for T1 delay
	TCA0.SPLIT.LCMP0 = T2_count_value;      // Set Counter B value for T2 delay
	
	TCA0.SPLIT.INTCTRL |= TCA0.SPLIT.HCMP0 | TCA0.SPLIT.LCMP0; // Enable interrupts for HCMP0 and LCMP0
}

// Delay function for T3 delay
void delay_T3() {
	TCA0.CMP0 = T3_count_value; // Set TCA.CMP0 for T3 delay
	while (!(TCA0.SPLIT.INTFLAGS & TCA0.SPLIT.LCMP0)); // Wait for T3 delay
}
